```vue
<script setup lang="ts">
import { ref, computed } from 'vue';

interface Props {
  title: string;
  visible?: boolean;
}

const props = withDefaults(defineProps<Props>(), {
  visible: true
});

const emit = defineEmits<{
  (e: 'update:visible', visible: boolean): void;
}>();

const displayTitle = computed(() => `Hello ${props.title}`);

const toggleVisibility = () => {
  emit('update:visible', !props.visible);
};
</script>

<template>
  <div v-if="visible" class="custom-component">
    <h2>{{ displayTitle }}</h2>
    <button @click="toggleVisibility">Toggle</button>
  </div>
</template>

<style scoped>
.custom-component {
  padding: 20px;
  border: 1px solid #ddd;
  border-radius: 4px;
}
</style>
```

## 5. API调用规范

### 5.1 API接口定义
```ts
// src/api/system/user.ts
import request from '@/utils/request';
import { UserQuery, UserInfo } from '@/types/user';

enum Api {
  UserList = '/system/user/list',
  GetUser = '/system/user/',
  AddUser = '/system/user',
  UpdateUser = '/system/user',
  DeleteUser = '/system/user/'
}

export const getUserList = (query: UserQuery) => {
  return request({
    url: Api.UserList,
    method: 'get',
    params: query
  });
};

export const getUserDetail = (userId: number | string) => {
  return request({
    url: Api.GetUser + userId,
    method: 'get'
  });
};
```

### 5.2 API调用示例
```ts
import { getUserList } from '@/api/system/user';
import { useMessage } from '@/hooks/web/useMessage';

const { createMessage } = useMessage();

const userList = ref<UserInfo[]>([]);
const loading = ref(false);

const fetchUserList = async (params: UserQuery) => {
  try {
    loading.value = true;
    const response = await getUserList(params);
    userList.value = response.data;
  } catch (error) {
    createMessage.error('获取用户列表失败');
  } finally {
    loading.value = false;
  }
};
```

## 6. 状态管理 (Pinia)

### 6.1 Store定义
```ts
// src/store/modules/user.ts
import { defineStore } from 'pinia';
import { getUserDetail } from '@/api/system/user';
import { UserInfo } from '@/types/user';

interface UserState {
  userInfo: Nullable<UserInfo>;
  token: string | undefined;
  permissions: string[];
}

export const useUserStore = defineStore({
  id: 'app-user',
  
  state: (): UserState => ({
    userInfo: null,
    token: undefined,
    permissions: []
  }),
  
  getters: {
    isLoggedIn(): boolean {
      return !!this.token;
    },
    
    hasPermission(): (permission: string) => boolean {
      return (permission: string) => this.permissions.includes(permission);
    }
  },
  
  actions: {
    async fetchUserInfo(userId: number) {
      try {
        const response = await getUserDetail(userId);
        this.userInfo = response.data;
        return Promise.resolve(response.data);
      } catch (error) {
        return Promise.reject(error);
      }
    },
    
    setToken(token: string) {
      this.token = token;
    },
    
    setPermissions(permissions: string[]) {
      this.permissions = permissions;
    }
  }
});
```

### 6.2 Store使用
```ts
// 在组件中使用
import { useUserStore } from '@/store/modules/user';

const userStore = useUserStore();

// 访问状态
console.log(userStore.userInfo);

// 调用Action
await userStore.fetchUserInfo(1);
```
